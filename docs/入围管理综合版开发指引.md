# 入围管理页面开发指引（综合版）

> 更新日期：2026-02-07  
> 页面：入围管理（综合书审+面谈，统一管理）  
> 角色：组委会/管理员

---

## 📋 页面功能概述

入围管理页面用于**综合管理入围决策**：

### 核心理念

- 在**一个页面**上同时展示所有项目的书审和面谈得分
- 支持**综合排名**（基于书审和面谈的加权得分）
- 支持**手动调整入围比例**（如前30%、前40%）
- 支持**个别增补**（手动标记某些项目入围或取消）

### 核心功能

1. ✅ 同时展示书审得分和面谈得分
2. ✅ 综合排名（可配置权重）
3. ✅ 按组别筛选
4. ✅ 手动调整入围比例（如30%、40%、50%）
5. ✅ 设置最低分数线
6. ✅ 个别增补/取消入围
7. ✅ 批量操作
8. ✅ 导出入围名单

---

## 🔌 API 策略

### 方案1：分别获取书审和面谈排名（推荐）

```javascript
// 1. 获取书审排名
const bookResp = await axios.get('/api/admin/reviews/rankings', {
  params: { competitionId: 21, stage: 'BOOK' }
});

// 2. 获取面谈排名
const interviewResp = await axios.get('/api/admin/reviews/rankings', {
  params: { competitionId: 21, stage: 'INTERVIEW' }
});

// 3. 前端合并数据
const mergedData = mergeScores(bookResp.data.data, interviewResp.data.data);
```

### 方案2：后端提供综合排名接口（如果有）

如果后端提供综合排名接口，可以直接调用。目前基于现有API，采用方案1。

### 页面特点

1. **双分数展示**：同时显示书审和面谈得分
2. **综合排名**：基于加权计算的综合得分排序
3. **状态标识**：
   - ✅ 入围（自动或手动）
   - ❌ 未入围
   - ⚠️ 待面谈（只有书审分数）
4. **灵活配置**：支持调整权重、比例、分数线
5. **个别增补**：可以手动标记任何项目入围

<script setup>
import { ref, computed, onMounted } from 'vue';
import axios from 'axios';
import { ElMessage, ElMessageBox } from 'element-plus';
import { Search } from '@element-plus/icons-vue';

// 配置
const config = ref({
  bookWeight: 50,
  interviewWeight: 50,
  ratioType: '30',
  customRatio: 30,
  minScore: null
});

// 筛选
const filters = ref({
  groupType: '',
  projectName: '',
  shortlistStatus: ''
});

// 数据
const projects = ref([]);
const loading = ref(false);

// 详情
const detailDialogVisible = ref(false);
const selectedProject = ref(null);
const bookDetail = ref(null);
const interviewDetail = ref(null);
const detailLoading = ref(false);
const activeTab = ref('BOOK');

// 计算属性
const filteredProjects = computed(() => {
  let list = projects.value;

  if (filters.value.groupType) {
    list = list.filter(p => p.groupType === filters.value.groupType);
  }

  if (filters.value.projectName) {
    const keyword = filters.value.projectName.toLowerCase();
    list = list.filter(p => p.projectName.toLowerCase().includes(keyword));
  }

  if (filters.value.shortlistStatus) {
    if (filters.value.shortlistStatus === 'shortlisted') {
      list = list.filter(p => p.isShortlisted);
    } else if (filters.value.shortlistStatus === 'not-shortlisted') {
      list = list.filter(p => !p.isShortlisted && p.compositeScore !== null);
    } else if (filters.value.shortlistStatus === 'pending-interview') {
      list = list.filter(p => p.compositeScore === null);
    }
  }

  return list;
});

const completedBookCount = computed(() => {
  return projects.value.filter(p => p.bookScore !== null).length;
});

const completedInterviewCount = computed(() => {
  return projects.value.filter(p => p.interviewScore !== null).length;
});

const shortlistedCount = computed(() => {
  return projects.value.filter(p => p.isShortlisted).length;
});

const shortlistRatio = computed(() => {
  if (projects.value.length === 0) return 0;
  return ((shortlistedCount.value / projects.value.length) * 100).toFixed(1);
});

const manualAddCount = computed(() => {
  return projects.value.filter(p => p.isManualAdd).length;
});

const manualRemoveCount = computed(() => {
  return projects.value.filter(p => p.isManualRemove).length;
});

// 方法
async function loadData() {
  loading.value = true;

  try {
    // 1. 获取书审排名
    const bookResp = await axios.get('/api/admin/reviews/rankings', {
      params: { competitionId: 21, stage: 'BOOK' },
      headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
    });

    // 2. 获取面谈排名
    const interviewResp = await axios.get('/api/admin/reviews/rankings', {
      params: { competitionId: 21, stage: 'INTERVIEW' },
      headers: { Authorization: `Bearer ${localStorage.getItem('token')}` }
    });

    // 3. 合并数据
    projects.value = mergeAndRank(bookResp.data.data, interviewResp.data.data);

    ElMessage.success(`加载成功，共 ${projects.value.length} 个项目`);
  } catch (error) {
    console.error('加载失败:', error);
    ElMessage.error('加载失败，请稍后重试');
  } finally {
    loading.value = false;
  }
}

function mergeAndRank(bookData, interviewData) {
  // 创建项目映射
  const projectMap = new Map();

  // 添加书审数据
  bookData.forEach(item => {
    projectMap.set(item.registrationId, {
      registrationId: item.registrationId,
      projectName: item.projectName,
      institutionName: item.institutionName,
      groupType: item.groupType,
      bookScore: item.avgTotal,
      interviewScore: null,
      compositeScore: null,
      isShortlisted: false,
      isManualAdd: false,
      isManualRemove: false,
      rank: null
    });
  });

  // 添加面谈数据
  interviewData.forEach(item => {
    if (projectMap.has(item.registrationId)) {
      const project = projectMap.get(item.registrationId);
      project.interviewScore = item.avgTotal;
    } else {
      // 只有面谈没有书审的情况（理论上不应该发生）
      projectMap.set(item.registrationId, {
        registrationId: item.registrationId,
        projectName: item.projectName,
        institutionName: item.institutionName,
        groupType: item.groupType,
        bookScore: null,
        interviewScore: item.avgTotal,
        compositeScore: null,
        isShortlisted: false,
        isManualAdd: false,
        isManualRemove: false,
        rank: null
      });
    }
  });

  // 计算综合得分
  const projects = Array.from(projectMap.values());
  projects.forEach(p => {
    if (p.bookScore !== null && p.interviewScore !== null) {
      p.compositeScore =
        (p.bookScore * config.value.bookWeight / 100) +
        (p.interviewScore * config.value.interviewWeight / 100);
    } else if (p.bookScore !== null && p.interviewScore === null) {
      // 只有书审，使用书审分数
      p.compositeScore = null; // 标记为待面谈
    }
  });

  // 排序（综合得分降序，待面谈的排在后面）
  projects.sort((a, b) => {
    if (a.compositeScore === null && b.compositeScore === null) return 0;
    if (a.compositeScore === null) return 1;
    if (b.compositeScore === null) return -1;
    return b.compositeScore - a.compositeScore;
  });

  // 分配排名
  projects.forEach((p, index) => {
    if (p.compositeScore !== null) {
      p.rank = index + 1;
    }
  });

  return projects;
}

function applyWeight() {
  if (config.value.bookWeight + config.value.interviewWeight !== 100) {
    ElMessage.warning('权重总和必须为100%');
    return;
  }

  // 重新计算综合得分和排名
  projects.value.forEach(p => {
    if (p.bookScore !== null && p.interviewScore !== null) {
      p.compositeScore =
        (p.bookScore * config.value.bookWeight / 100) +
        (p.interviewScore * config.value.interviewWeight / 100);
    }
  });

  // 重新排序
  projects.value.sort((a, b) => {
    if (a.compositeScore === null && b.compositeScore === null) return 0;
    if (a.compositeScore === null) return 1;
    if (b.compositeScore === null) return -1;
    return b.compositeScore - a.compositeScore;
  });

  // 重新分配排名
  projects.value.forEach((p, index) => {
    if (p.compositeScore !== null) {
      p.rank = index + 1;
    }
  });

  ElMessage.success('权重已应用，排名已更新');
}

function applyRatio() {
  const ratio = config.value.ratioType === 'custom'
    ? config.value.customRatio
    : parseInt(config.value.ratioType);

  const eligibleProjects = projects.value.filter(p => p.compositeScore !== null);
  const shortlistCount = Math.ceil(eligibleProjects.length * ratio / 100);

  ElMessageBox.confirm(
    `将按照${ratio}%的比例设置入围，共${shortlistCount}个项目。是否继续？`,
    '确认',
    {
      confirmButtonText: '确定',
      cancelButtonText: '取消',
      type: 'info'
    }
  ).then(() => {
    // 重置所有入围状态
    projects.value.forEach(p => {
      p.isShortlisted = false;
      p.isManualAdd = false;
      p.isManualRemove = false;
    });

    // 设置前N名入围
    eligibleProjects.slice(0, shortlistCount).forEach(p => {
      p.isShortlisted = true;
    });

    ElMessage.success(`已设置前${shortlistCount}个项目入围`);
  }).catch(() => {
    ElMessage.info('已取消');
  });
}

  ElMessage.success('导出成功');
}
