# 报名统计 - 品管工具分布使用指引

## 更新说明

✅ **已添加品管工具分布字段 `methodCounts`**

统计接口现在返回3个分布数据：
1. `regionCounts` - 地区分布
2. `subjectTypeCounts` - 主题类型分布
3. **`methodCounts`** - 品管工具分布 ⭐ 新增

## API信息

**接口路径**: `GET /api/admin/stats/summary`

**权限**: 组委会管理员 (`COMMITTEE_ADMIN`)

**参数**:
- `competitionId` (可选): 指定赛事ID，不传则使用最新赛事

## 前端使用示例

### Vue.js 示例

```vue
<template>
  <div class="stats-dashboard">
    <!-- 品管工具分布图表 -->
    <el-card title="品管工具分布">
      <div ref="methodChart" style="width: 100%; height: 400px;"></div>
    </el-card>
    
    <!-- 品管工具分布表格 -->
    <el-table :data="methodTableData" border>
      <el-table-column prop="method" label="品管工具" width="200"></el-table-column>
      <el-table-column prop="count" label="数量" width="100"></el-table-column>
      <el-table-column prop="percentage" label="占比" width="100">
        <template #default="scope">
          {{ scope.row.percentage }}%
        </template>
      </el-table-column>
    </el-table>
  </div>
</template>

<script>
import * as echarts from 'echarts';

export default {
  data() {
    return {
      statsData: null,
      methodTableData: []
    }
  },
  methods: {
    async fetchStats() {
      const response = await fetch('/api/admin/stats/summary', {
        headers: {
          'Authorization': `Bearer ${this.$store.state.token}`
        }
      });
      
      const result = await response.json();
      if (result.success) {
        this.statsData = result.data;
        this.processMethodData();
        this.renderMethodChart();
      }
    },
    
    processMethodData() {
      const methodCounts = this.statsData.methodCounts || {};
      const total = Object.values(methodCounts).reduce((sum, count) => sum + count, 0);
      
      this.methodTableData = Object.entries(methodCounts)
        .map(([method, count]) => ({
          method,
          count,
          percentage: ((count / total) * 100).toFixed(1)
        }))
        .sort((a, b) => b.count - a.count);
    },
    
    renderMethodChart() {
      const chart = echarts.init(this.$refs.methodChart);
      const methodCounts = this.statsData.methodCounts || {};
      
      const option = {
        title: {
          text: '品管工具分布',
          left: 'center'
        },
        tooltip: {
          trigger: 'item',
          formatter: '{b}: {c} ({d}%)'
        },
        legend: {
          orient: 'vertical',
          left: 'left'
        },
        series: [
          {
            name: '品管工具',
            type: 'pie',
            radius: '50%',
            data: Object.entries(methodCounts).map(([name, value]) => ({
              name,
              value
            })),
            emphasis: {
              itemStyle: {
                shadowBlur: 10,
                shadowOffsetX: 0,
                shadowColor: 'rgba(0, 0, 0, 0.5)'
              }
            }
          }
        ]
      };
      
      chart.setOption(option);
    }
  },
  mounted() {
    this.fetchStats();
  }
}
</script>
```

### React 示例

```jsx
import React, { useState, useEffect, useRef } from 'react';
import * as echarts from 'echarts';

function StatsMethodDistribution() {
  const [methodCounts, setMethodCounts] = useState({});
  const chartRef = useRef(null);

  useEffect(() => {
    fetchStats();
  }, []);

  const fetchStats = async () => {
    const response = await fetch('/api/admin/stats/summary', {
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('token')}`
      }
    });
    
    const result = await response.json();
    if (result.success) {
      const counts = result.data.methodCounts || {};
      setMethodCounts(counts);
      renderChart(counts);
    }
  };

  const renderChart = (counts) => {
    const chart = echarts.init(chartRef.current);
    
    const option = {
      title: {
        text: '品管工具分布',
        left: 'center'
      },
      tooltip: {
        trigger: 'item',
        formatter: '{b}: {c} ({d}%)'
      },
      series: [
        {
          name: '品管工具',
          type: 'pie',
          radius: '50%',
          data: Object.entries(counts).map(([name, value]) => ({
            name,
            value
          }))
        }
      ]
    };
    
    chart.setOption(option);
  };

  const total = Object.values(methodCounts).reduce((sum, count) => sum + count, 0);

  return (
    <div>
      <h2>品管工具分布</h2>
      <div ref={chartRef} style={{ width: '100%', height: '400px' }}></div>
      
      <table>
        <thead>
          <tr>
            <th>品管工具</th>
            <th>数量</th>
            <th>占比</th>
          </tr>
        </thead>
        <tbody>
          {Object.entries(methodCounts)
            .sort((a, b) => b[1] - a[1])
            .map(([method, count]) => (
              <tr key={method}>
                <td>{method}</td>
                <td>{count}</td>
                <td>{((count / total) * 100).toFixed(1)}%</td>
              </tr>
            ))}
        </tbody>
      </table>
    </div>
  );
}

export default StatsMethodDistribution;
```

### 原生JavaScript示例

```javascript
// 获取统计数据
async function fetchStats() {
  const response = await fetch('/api/admin/stats/summary', {
    headers: {
      'Authorization': `Bearer ${getToken()}`
    }
  });
  
  const result = await response.json();
  if (result.success) {
    displayMethodDistribution(result.data.methodCounts);
  }
}

// 显示品管工具分布
function displayMethodDistribution(methodCounts) {
  const total = Object.values(methodCounts).reduce((sum, count) => sum + count, 0);
  
  // 渲染表格
  const tbody = document.querySelector('#method-table tbody');
  tbody.innerHTML = '';
  
  Object.entries(methodCounts)
    .sort((a, b) => b[1] - a[1])
    .forEach(([method, count]) => {
      const percentage = ((count / total) * 100).toFixed(1);
      const row = `
        <tr>
          <td>${method}</td>
          <td>${count}</td>
          <td>${percentage}%</td>
        </tr>
      `;
      tbody.innerHTML += row;
    });
  
  // 渲染图表
  renderMethodChart(methodCounts);
}

// 渲染ECharts图表
function renderMethodChart(methodCounts) {
  const chart = echarts.init(document.getElementById('method-chart'));
  
  const option = {
    title: {
      text: '品管工具分布',
      left: 'center'
    },
    tooltip: {
      trigger: 'item',
      formatter: '{b}: {c} ({d}%)'
    },
    series: [
      {
        name: '品管工具',
        type: 'pie',
        radius: '50%',
        data: Object.entries(methodCounts).map(([name, value]) => ({
          name,
          value
        }))
      }
    ]
  };

}
```

## 数据处理技巧

### 计算占比
```javascript
const total = Object.values(methodCounts).reduce((sum, count) => sum + count, 0);
const percentage = (count / total * 100).toFixed(1);
```

### 排序（按数量降序）
```javascript
const sorted = Object.entries(methodCounts)
  .sort((a, b) => b[1] - a[1]);
```

### 过滤小占比项
```javascript
const threshold = 0.05; // 5%
const filtered = Object.entries(methodCounts)
  .filter(([_, count]) => count / total >= threshold);
```

### 合并"其他"项
```javascript
const threshold = 0.05;
const total = Object.values(methodCounts).reduce((sum, count) => sum + count, 0);
const result = {};
let otherCount = 0;

Object.entries(methodCounts).forEach(([method, count]) => {
  if (count / total >= threshold) {
    result[method] = count;
  } else {
    otherCount += count;
  }
});

if (otherCount > 0) {
  result['其他'] = otherCount;
}
```

## 注意事项

1. **字段名称**: `methodCounts` (驼峰命名)
2. **数据类型**: `Map<String, Integer>` (品管工具Label → 数量)
3. **Label显示**: 返回的是中文Label，不是code
4. **未知处理**: 如果没有填写品管工具，会显示为"未知"
5. **权限**: 只有组委会管理员可以访问此接口
