# 面谈得分列表 - 前端开发指引

## 一、功能概述

面谈得分列表用于展示面谈阶段的评审得分，页面样式与书审得分列表完全一致。

## 二、API接口

### 2.1 接口地址
```
GET /api/admin/reviews/interview-scores
```

### 2.2 请求参数

| 参数名 | 类型 | 必填 | 说明 | 示例 |
|--------|------|------|------|------|
| competitionId | Long | 是 | 竞赛ID | 21 |
| status | String | 否 | 任务状态 | SCORED |
| reviewerId | Long | 否 | 评委ID | 84 |
| institutionId | Long | 否 | 机构ID | 1 |
| groupType | String | 否 | 组别类型 | ADVANCED |

**status可选值**：
- `PENDING` - 待确认
- `CONFIRMED` - 已确认
- `SCORED` - 已评分
- `RETURNED` - 已驳回

**groupType可选值**：
- `BASIC` - 基层组
- `ADVANCED` - 进阶组

### 2.3 请求示例

```javascript
// 基础查询
axios.get('/api/admin/reviews/interview-scores', {
  params: {
    competitionId: 21
  }
})

// 带筛选条件
axios.get('/api/admin/reviews/interview-scores', {
  params: {
    competitionId: 21,
    status: 'SCORED',
    groupType: 'ADVANCED'
  }
})
```

### 3.2 Vue组件示例

```vue
<template>
  <div class="score-list-container">
    <!-- Tab切换 -->
    <el-tabs v-model="activeTab" @tab-click="handleTabChange">
      <el-tab-pane label="书审得分" name="book"></el-tab-pane>
      <el-tab-pane label="面谈得分" name="interview"></el-tab-pane>
    </el-tabs>

    <!-- 筛选条件 -->
    <el-form :inline="true" :model="queryParams" class="filter-form">
      <el-form-item label="状态">
        <el-select v-model="queryParams.status" placeholder="全部" clearable>

      <el-table-column prop="plan" label="计划" width="70" />
      <el-table-column prop="problem" label="问题" width="70" />
      <el-table-column prop="action" label="行动" width="70" />
      <el-table-column prop="success" label="成功" width="70" />

      <el-table-column prop="submittedAt" label="提交时间" width="160" />

      v-if="total > 0"
      :current-page="queryParams.page"

      @current-change="handlePageChange"
    />

<script>
import { getBookScores, getInterviewScores } from '@/api/review'

export default {
  name: 'ScoreList',
  data() {
    return {
      activeTab: 'book',
      loading: false,
      scoreList: [],
      total: 0,
      queryParams: {
        competitionId: 21,
        status: '',
        groupType: '',
        page: 1,
        pageSize: 20
      }
    }
  },
  created() {
    this.fetchScores()
  },
  methods: {
    // 切换Tab
    handleTabChange(tab) {
      this.activeTab = tab.name
      this.resetQuery()
      this.fetchScores()
    },
    
    // 获取得分列表
    async fetchScores() {
      this.loading = true
      try {
        const apiMethod = this.activeTab === 'book' 
          ? getBookScores 
          : getInterviewScores
        
        const { data } = await apiMethod(this.queryParams)
        this.scoreList = data
        this.total = data.length
      } catch (error) {
        this.$message.error('获取得分列表失败')
      } finally {
        this.loading = false
      }
    },
    
    // 重置查询
    resetQuery() {
      this.queryParams = {
        competitionId: 21,
        status: '',
        groupType: '',
        page: 1,
        pageSize: 20
      }
      this.fetchScores()
    },
    
    // 分页
    handlePageChange(page) {
      this.queryParams.page = page
      this.fetchScores()
    },
    
    // 查看详情
    viewDetail(row) {
      this.$router.push({
        name: 'ScoreDetail',
        params: { scoreId: row.scoreId }
      })
    },
    
    // 导出
    exportScore(row) {
      // 导出逻辑
    }
  }
}
</script>
```

### 3.3 React组件示例

```jsx
import React, { useState, useEffect } from 'react';
import { Table, Tabs, Form, Select, Button, message } from 'antd';
import { getBookScores, getInterviewScores } from '@/api/review';

const { TabPane } = Tabs;
const { Option } = Select;

const ScoreList = () => {
  const [activeTab, setActiveTab] = useState('book');
  const [loading, setLoading] = useState(false);
  const [scoreList, setScoreList] = useState([]);
  const [queryParams, setQueryParams] = useState({
    competitionId: 21,
    status: '',
    groupType: ''
  });

  useEffect(() => {
    fetchScores();
  }, [activeTab]);

  const fetchScores = async () => {
    setLoading(true);
    try {
      const apiMethod = activeTab === 'book' ? getBookScores : getInterviewScores;
      const { data } = await apiMethod(queryParams);
      setScoreList(data);
    } catch (error) {
      message.error('获取得分列表失败');
    } finally {
      setLoading(false);
    }
  };

  const columns = [
    { title: '项目名称', dataIndex: 'projectName', width: 200 },
    { title: '机构', dataIndex: 'institutionName', width: 150 },
    { title: '分组', dataIndex: 'groupCode', width: 80 },
    { title: '评委', dataIndex: 'reviewerName', width: 100 },
    { title: '计划', dataIndex: 'plan', width: 70 },
    { title: '问题', dataIndex: 'problem', width: 70 },
    { title: '行动', dataIndex: 'action', width: 70 },
    { title: '成功', dataIndex: 'success', width: 70 },
    { title: '评审', dataIndex: 'review', width: 70 },
    { title: '操作', dataIndex: 'operation', width: 70 },
    { title: '展示', dataIndex: 'presentation', width: 70 },
    {
      title: '总分',
      dataIndex: 'total',
      width: 80,
      render: (text) => <span style={{ fontWeight: 'bold', color: '#1890ff' }}>{text}</span>
    },
    { title: '提交时间', dataIndex: 'submittedAt', width: 160 }
  ];

  return (
    <div className="score-list-container">
      <Tabs activeKey={activeTab} onChange={setActiveTab}>
        <TabPane tab="书审得分" key="book" />
        <TabPane tab="面谈得分" key="interview" />
      </Tabs>

      <Form layout="inline" style={{ marginBottom: 16 }}>
        <Form.Item label="状态">
          <Select
            value={queryParams.status}
            onChange={(value) => setQueryParams({ ...queryParams, status: value })}
            style={{ width: 120 }}
            allowClear
          >
            <Option value="SCORED">已评分</Option>
            <Option value="RETURNED">已驳回</Option>
          </Select>
        </Form.Item>
        
        <Form.Item label="组别">
          <Select
            value={queryParams.groupType}
            onChange={(value) => setQueryParams({ ...queryParams, groupType: value })}
            style={{ width: 120 }}
            allowClear
          >
            <Option value="BASIC">基层组</Option>
            <Option value="ADVANCED">进阶组</Option>
          </Select>
        </Form.Item>
        
        <Form.Item>
          <Button type="primary" onClick={fetchScores}>查询</Button>
        </Form.Item>
      </Form>

      <Table
        columns={columns}
        dataSource={scoreList}
        loading={loading}
        rowKey="scoreId"
        bordered
        scroll={{ x: 1500 }}
      />
    </div>
  );
};

export default ScoreList;
```

## 四、关键注意事项

### 4.1 数据结构完全相同
书审和面谈的数据结构完全一致，可以复用同一套组件和样式。

### 4.2 面谈特点
- 面谈通常只有进阶组（ADVANCED）项目
- 面谈是在书审之后的阶段
- 评分标准与书审相同（7项指标）

### 4.3 筛选建议
```javascript
// 面谈得分列表默认筛选进阶组

  competitionId: 21,
  groupType: 'ADVANCED',  // 面谈通常是进阶组

};
```
