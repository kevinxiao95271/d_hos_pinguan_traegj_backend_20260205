### 1. 基本调用

```vue
<template>
  <div class="registration-filter">
    <!-- 品管工具选择器 -->
    <el-select
      v-model="selectedMethod"
      placeholder="选择品管工具"
      @change="filterRegistrations"
      clearable
    >
      <el-option label="品管圈-课题达成" value="品管圈-课题达成" />
      <el-option label="PDCA" value="PDCA" />
      <el-option label="根本原因分析" value="根本原因分析" />
      <el-option label="5S" value="5S" />
      <!-- 更多选项 -->
    </el-select>

    <!-- 数据表格 -->
    <el-table :data="registrations" border v-loading="loading">
      <el-table-column prop="registrationId" label="项目编号" width="100" />
      <el-table-column prop="projectName" label="项目名称" min-width="200" />
      <el-table-column prop="institutionName" label="医疗机构名称" min-width="200" />
      <el-table-column prop="groupType" label="竞赛组别" width="120">
        <template #default="{ row }">
          <span v-if="row.groupType === 'BASIC'">基层组</span>
          <span v-else-if="row.groupType === 'ADVANCED'">进阶组</span>
          <span v-else-if="row.groupType === 'COMPREHENSIVE'">综合组</span>
          <span v-else>{{ row.groupType }}</span>
        </template>
      </el-table-column>
      <el-table-column prop="groupCode" label="分组" width="100" />
      <el-table-column prop="methodLabel" label="品管工具" width="150" />
      <el-table-column prop="applicantName" label="报名人" width="120" />
      <el-table-column prop="submittedAt" label="报名时间" width="180">
        <template #default="{ row }">
          {{ formatDate(row.submittedAt) }}
        </template>
      </el-table-column>
      <el-table-column label="操作" width="150">
        <template #default="{ row }">
          <el-button size="small" @click="viewDetail(row.registrationId)">
            查看详情
          </el-button>
        </template>
      </el-table-column>
    </el-table>

    <!-- 空状态 -->
    <el-empty v-if="!loading && registrations.length === 0" description="暂无报名数据">
      <template #description>
        <p>暂无报名数据。请确保：</p>
        <p>1) 已创建赛事</p>
        <p>2) 有参赛者报名</p>
        <p>3) 筛选条件正确</p>
      </template>
    </el-empty>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import axios from 'axios';
import { ElMessage } from 'element-plus';

const selectedMethod = ref('');
const registrations = ref([]);
const loading = ref(false);

// 筛选报名
async function filterRegistrations() {
  loading.value = true;
  
  try {
    const params = {
      competitionId: 21  // 当前赛事ID
    };
    
    // 如果选择了品管工具，添加筛选条件
    if (selectedMethod.value) {
      params.methodLabel = selectedMethod.value;
    }
    
    const response = await axios.get('/api/admin/registrations/filter', {
      params: params,
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });
    
    registrations.value = response.data.data;
    
    // 显示结果
    if (registrations.value.length === 0) {
      ElMessage.warning(`未找到符合条件的报名数据`);
    } else {
      ElMessage.success(`找到 ${registrations.value.length} 条报名`);
    }
  } catch (error) {
    console.error('筛选失败:', error);
    ElMessage.error('筛选失败，请稍后重试');
    registrations.value = [];
  } finally {
    loading.value = false;
  }
}

// 格式化日期
function formatDate(dateStr) {
  if (!dateStr) return '-';
  const date = new Date(dateStr);
  return date.toLocaleString('zh-CN', {
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    hour: '2-digit',
    minute: '2-digit'
  });
}

// 查看详情
function viewDetail(registrationId) {
  // 跳转到详情页或打开详情对话框
  console.log('查看详情:', registrationId);
}

// 初始加载
onMounted(() => {
  filterRegistrations();
});
</script>
```

import { Table, Select, Spin, Empty, message } from 'antd';

const RegistrationFilter = () => {

  const [registrations, setRegistrations] = useState([]);

  // 品管工具选项
  const methodOptions = [

    {

      dataIndex: 'registrationId',
      key: 'registrationId',
      width: 100,
    },
    {
      title: '项目名称',

      width: 200,
    },

      title: '医疗机构名称',
      dataIndex: 'institutionName',

        return map[groupType] || groupType;
      }

    {
      title: '分组',
      dataIndex: 'groupCode',
      key: 'groupCode',
      width: 100,
    },
    {
      title: '品管工具',
      dataIndex: 'methodLabel',
      key: 'methodLabel',
      width: 150,
    },
    {
      title: '报名人',
      dataIndex: 'applicantName',
      key: 'applicantName',
      width: 120,
    },
    {
      title: '报名时间',
      dataIndex: 'submittedAt',
      key: 'submittedAt',
      width: 180,
      render: (date) => {
        if (!date) return '-';
        return new Date(date).toLocaleString('zh-CN');
      }
    },
    {
      title: '操作',
      key: 'action',
      width: 150,

        <button onClick={() => viewDetail(record.registrationId)}>
          查看详情

    }

  const filterRegistrations = async (methodLabel) => {
    setLoading(true);
    
    try {
      const params = {
        competitionId: 21
      };
      
      if (methodLabel) {
        params.methodLabel = methodLabel;
      }
      
      const response = await axios.get('/api/admin/registrations/filter', {
        params: params,

        message.success(`找到 ${response.data.data.length} 条报名`);

      setLoading(false);

          )

  );

## 🔍 故障排查

### 问题1：返回空数据

**可能原因**：
1. ❌ API路径错误
   - 错误：`/api/registrations/filter`
   - 正确：`/api/admin/registrations/filter`

2. ❌ 赛事ID错误
   - 确认当前赛事ID是否为21

3. ❌ label拼写错误
   - 确认是"品管圈-课题达成"（使用中文破折号"-"，不是英文"-"）

4. ❌ token失效
   - 检查localStorage中的token是否存在
   - 重新登录获取新token

5. ❌ 参数未正确传递
   - 检查浏览器Network面板，确认URL参数正确

### 问题2：返回401未授权

**解决方案**：
```javascript
// 确保请求头包含token
headers: {
  Authorization: `Bearer ${localStorage.getItem('token')}`
}
```

### 问题3：返回500服务器错误

**解决方案**：
1. 检查后端服务器是否正常运行
2. 查看服务器日志
3. 确认数据库连接正常
