## 🎯 前端实现方案

### 方案1: 任务列表页 → 评审详情页（推荐）

#### Step 1: 任务列表页 (`ReviewerTaskList.vue`)

```vue
<template>
  <div class="task-list">
    <el-table :data="tasks" @row-click="handleTaskClick">
      <el-table-column prop="projectName" label="项目名称" />
      <el-table-column prop="institutionName" label="机构" />
      <el-table-column prop="stage" label="阶段" />
      <el-table-column prop="status" label="状态" />
      <el-table-column label="操作">
        <template #default="{ row }">
          <el-button @click="goToReview(row)">评审</el-button>
        </template>
      </el-table-column>
    </el-table>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import { useRouter } from 'vue-router';
import axios from 'axios';

const router = useRouter();
const tasks = ref([]);

// 获取任务列表
const fetchTasks = async () => {
  try {
    const response = await axios.get('/api/reviews/my-tasks', {
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });
    
    // ⭐ 关键：直接使用返回的数据，包含 registrationId 和 projectName
    tasks.value = response.data.data;
    
    console.log('任务列表:', tasks.value); // 调试用
  } catch (error) {
    console.error('获取任务失败:', error);
  }
};

// 跳转到评审页面
const goToReview = (task) => {
  // ⭐ 方法1：通过路由参数传递（推荐）
  router.push({
    name: 'ReviewDetail',
    params: { 
      taskId: task.id 
    },
    query: { 
      registrationId: task.registrationId  // ⭐ 传递 registrationId
    }
  });
  
  // ⭐ 方法2：通过路由参数传递两个ID
  // router.push(`/reviewer/review/${task.id}/${task.registrationId}`);
};

onMounted(() => {
  fetchTasks();
});
</script>
```

#### Step 2: 路由配置 (`router/index.js`)

```javascript
const routes = [
  {
    path: '/reviewer',
    component: ReviewerLayout,
    children: [
      {
        path: 'tasks',
        name: 'ReviewerTaskList',
        component: () => import('@/views/reviewer/TaskList.vue')
      },
      {
        path: 'review/:taskId',
        name: 'ReviewDetail',
        component: () => import('@/views/reviewer/ReviewDetail.vue'),
        // ⭐ 可以在路由元信息中标记需要的参数
        meta: {
          requiresAuth: true,
          role: 'REVIEWER'
        }
      }
    ]
  }
];
```

#### Step 3: 评审详情页 (`ReviewDetail.vue`)

```vue
<template>
  <div class="review-detail" v-if="registration">
    <!-- 项目基本信息 -->
    <el-card class="project-info">
      <h2>{{ registration.projectName }}</h2>
      <p><strong>机构:</strong> {{ institution?.name }}</p>
      <p><strong>组别:</strong> {{ registration.groupType }}</p>
      <p><strong>状态:</strong> {{ registration.status }}</p>
    </el-card>

    <!-- 成员信息 -->
    <el-card class="members">
      <h3>成员列表</h3>
      <el-table :data="members">
        <el-table-column prop="name" label="姓名" />
        <el-table-column prop="title" label="职称" />
        <el-table-column prop="role" label="角色" />
        <el-table-column prop="department" label="部门" />
      </el-table>
    </el-card>

    <!-- 活动说明 -->
    <el-card class="activity" v-if="activityInfo">
      <h3>活动说明</h3>
      <p><strong>主题:</strong> {{ activityInfo.theme }}</p>
      <p><strong>关键词:</strong> {{ activityInfo.keywords }}</p>
      <!-- 其他字段... -->
    </el-card>

    <!-- 项目总结 -->
    <el-card class="summary" v-if="summary">
      <h3>项目总结</h3>
      <div v-html="summary.conclusion"></div>
    </el-card>

    <!-- 评分表单 -->
    <el-card class="score-form">
      <h3>评审打分</h3>
      <el-form :model="scoreForm" label-width="120px">
        <el-form-item label="计划 (Plan)">
          <el-input-number v-model="scoreForm.plan" :min="0" :max="20" />
        </el-form-item>
        <el-form-item label="问题 (Problem)">
          <el-input-number v-model="scoreForm.problem" :min="0" :max="20" />
        </el-form-item>
        <el-form-item label="措施 (Action)">
          <el-input-number v-model="scoreForm.action" :min="0" :max="20" />
        </el-form-item>
        <el-form-item label="成效 (Success)">
          <el-input-number v-model="scoreForm.success" :min="0" :max="20" />
        </el-form-item>
        <el-form-item label="回顾 (Review)">
          <el-input-number v-model="scoreForm.review" :min="0" :max="20" />
        </el-form-item>
        <el-form-item label="亮点">
          <el-input type="textarea" v-model="scoreForm.highlight" />
        </el-form-item>
        <el-form-item label="不足">
          <el-input type="textarea" v-model="scoreForm.weakness" />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="submitScore">提交评分</el-button>
        </el-form-item>
      </el-form>
    </el-card>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue';
import { useRoute } from 'vue-router';
import axios from 'axios';
import { ElMessage } from 'element-plus';

const route = useRoute();

// ⭐ 从路由参数获取 taskId 和 registrationId
const taskId = ref(route.params.taskId);
const registrationId = ref(route.query.registrationId);

const registration = ref(null);
const institution = ref(null);
const members = ref([]);
const activityInfo = ref(null);
const summary = ref(null);

const scoreForm = ref({
  plan: 0,
  problem: 0,
  action: 0,
  success: 0,
  review: 0,
  operation: 0,
  presentation: 0,
  highlight: '',
  weakness: ''
});

// 加载报名详情
const fetchRegistrationDetail = async () => {
  // ⭐ 检查 registrationId 是否存在
  if (!registrationId.value) {
    ElMessage.error('缺少项目ID，无法加载详情');
    console.error('registrationId is missing:', route.query);
    return;
  }

  try {
    const response = await axios.get(`/api/registrations/${registrationId.value}`, {
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });
    
    const data = response.data.data;
    
    // ⭐ 解析返回的数据结构
    registration.value = data.registration;
    institution.value = data.institution;
    members.value = data.members || [];
    activityInfo.value = data.activityInfo;
    summary.value = data.summary;
    
    console.log('报名详情:', data); // 调试用
  } catch (error) {
    console.error('加载报名详情失败:', error);
    ElMessage.error('加载项目详情失败');
  }
};

// 提交评分
const submitScore = async () => {
  try {
    await axios.post('/api/reviews/scores', {
      reviewTaskId: taskId.value,  // ⭐ 使用 taskId，不是 registrationId
      ...scoreForm.value
    }, {
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });
    
    ElMessage.success('评分提交成功');
    // 跳转回任务列表
    // router.push('/reviewer/tasks');
  } catch (error) {
    console.error('提交评分失败:', error);
    ElMessage.error('提交评分失败');
  }
};

onMounted(() => {
  console.log('Route params:', route.params);  // 调试用
  console.log('Route query:', route.query);    // 调试用
  fetchRegistrationDetail();
});
</script>
```
