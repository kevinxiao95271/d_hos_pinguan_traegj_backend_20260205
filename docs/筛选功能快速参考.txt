async function filterByMethod(methodLabel) {
  const response = await axios.get('/api/admin/registrations/filter', {
    params: {
      competitionId: 21,
      methodLabel: methodLabel
    },
    headers: {
      Authorization: `Bearer ${localStorage.getItem('token')}`
    }
  });
  
  return response.data.data;
}

// 调用

console.log(results);  // 应该有2个报名

================================================================================
  常见问题
================================================================================

问题：返回空数据 []

检查清单：
  ☐ API路径：/api/admin/registrations/filter （不是/api/registrations/filter）
  ☐ 赛事ID：competitionId=21
  ☐ label拼写：methodLabel=品管圈-课题达成
  ☐ token传递：Authorization: Bearer YOUR_TOKEN
  ☐ 浏览器Network面板：状态码200？
  ☐ Response data：是[]还是有数据？

问题：返回401

解决：检查token是否正确传递
  headers: {
    Authorization: `Bearer ${localStorage.getItem('token')}`
  }

问题：返回500

解决：检查后端服务器是否正常运行
