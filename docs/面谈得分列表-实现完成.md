### 组件复用
由于数据结构完全相同，前端可以复用书审得分列表的组件：

```vue
<template>
  <div>
    <!-- Tab切换 -->
    <el-tabs v-model="activeTab" @tab-click="handleTabClick">
      <el-tab-pane label="书审得分" name="book"></el-tab-pane>
      <el-tab-pane label="面谈得分" name="interview"></el-tab-pane>
    </el-tabs>
    
    <!-- 复用同一个表格组件 -->
    <ScoreTable :scores="scores" :loading="loading" />
  </div>
</template>

<script>
export default {
  data() {
    return {
      activeTab: 'book',
      scores: [],
      loading: false
    };
  },
  methods: {
    async handleTabClick(tab) {
      this.loading = true;
      
      const endpoint = tab.name === 'book' 
        ? '/api/admin/reviews/book-scores'
        : '/api/admin/reviews/interview-scores';
      
      const response = await axios.get(endpoint, {
        params: { competitionId: this.competitionId }
      });
      
      this.scores = response.data.data;
      this.loading = false;
    }
  }
};
</script>
```
