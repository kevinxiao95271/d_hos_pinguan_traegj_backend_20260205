# 入围管理页面前端开发指引（最终版）

> 更新日期：2026-02-07  
> 页面：入围管理（书审入围 + 面谈入围）  
> 角色：组委会/管理员

---

## 📋 页面功能概述

入围管理页面用于管理**两个入围阶段**：

### 两个入围阶段

1. **书审入围**：查看书审得分，筛选入围面谈的项目
2. **面谈入围**：查看面谈得分，筛选入围决赛的项目

> **注意**：决赛不属于"入围"范畴，决赛是最终比赛阶段，其结果管理应该在其他页面（如成绩管理、结果公示）。

### 核心功能

1. ✅ 切换入围阶段（书审入围/面谈入围）
2. ✅ 查看该阶段得分排名
3. ✅ 按组别筛选
4. ✅ 设置入围分数线
5. ✅ 设置入围数量限制
6. ✅ 查看项目详情
7. ✅ 导出入围名单

## 🔌 API 端点

### 1. 获取评分排名

```
GET /api/admin/reviews/rankings
```

**参数：**
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| competitionId | Long | ✅ | 赛事ID |
| stage | String | ✅ | **BOOK（书审）或 INTERVIEW（面谈）** |
| groupType | String | ❌ | 组别筛选（BASIC/ADVANCED/COMPREHENSIVE） |

**返回数据：**
```json
{
  "code": 200,
  "message": "success",
  "data": [
    {
      "rank": 1,
      "registrationId": 106,
      "projectName": "护理交接班规范化-1",
      "institutionName": "浙江大学医学院附属第一医院",
      "groupType": "BASIC",
      "stage": "BOOK",  // 或 "INTERVIEW"
      "avgTotal": 92.5
    }
  ]
}
```

### 2. 获取入围名单（带筛选）

```
GET /api/admin/reviews/shortlist
```

**参数：**
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| competitionId | Long | ✅ | 赛事ID |
| stage | String | ✅ | **BOOK（书审）或 INTERVIEW（面谈）** |
| groupType | String | ❌ | 组别筛选 |
| limit | Integer | ❌ | 入围数量限制（默认10） |
| minAvgTotal | Double | ❌ | 最低分数线 |

### 3. 获取项目详细评分

```
GET /api/registrations/{id}/review-details
```

**返回数据：**
```json
{
  "code": 200,
  "message": "success",
  "data": [
    {
      "stage": "BOOK",
      "avgPlan": 18.5,
      "avgProblem": 19.0,
      "avgAction": 18.8,
      "avgSuccess": 14.2,
      "avgReview": 9.5,
      "avgOperation": 8.0,
      "avgPresentation": 4.5,
      "avgTotal": 92.5,
      "highlights": ["亮点1", "亮点2"],
      "weaknesses": ["不足1", "不足2"]
    },
    {
      "stage": "INTERVIEW",
      "avgPlan": 19.0,
      "avgProblem": 18.5,
      // ... 面谈阶段评分
      "avgTotal": 90.3,
      "highlights": ["面谈表现优秀"],
      "weaknesses": ["回答问题可以更详细"]
    }
  ]
}
```

<script setup>
import { ref, computed, onMounted } from 'vue';
import axios from 'axios';
import { ElMessage } from 'element-plus';

// 当前入围阶段（只有书审和面谈）
const currentStage = ref('BOOK');

// 筛选条件
const filters = ref({
  groupType: '',
  minScore: null,
  limit: 10
});

// 数据
const allRankings = ref([]);
const loading = ref(false);
const shortlistedIds = ref(new Set());

// 详情对话框
const detailDialogVisible = ref(false);
const selectedDetail = ref(null);
const scoreDetails = ref([]);
const detailLoading = ref(false);
const activeTab = ref('BOOK');

// 计算属性
const filteredList = computed(() => {
  let list = allRankings.value;

  if (filters.value.minScore) {
    list = list.filter(item => item.avgTotal >= filters.value.minScore);
  }

  if (filters.value.limit) {
    list = list.slice(0, filters.value.limit);
  }

  return list;
});

const averageScore = computed(() => {
  if (allRankings.value.length === 0) return '0.0';
  const sum = allRankings.value.reduce((acc, item) => acc + item.avgTotal, 0);
  return (sum / allRankings.value.length).toFixed(1);
});

const maxScore = computed(() => {
  if (allRankings.value.length === 0) return '0.0';
  return Math.max(...allRankings.value.map(item => item.avgTotal)).toFixed(1);
});

const shortlistedCount = computed(() => {
  return filteredList.value.filter(item => isShortlisted(item)).length;
});

// 方法
async function loadRankings() {
  loading.value = true;

  try {
    const response = await axios.get('/api/admin/reviews/rankings', {
      params: {
        competitionId: 21,
        stage: currentStage.value,
        groupType: filters.value.groupType || undefined
      },
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });

    allRankings.value = response.data.data;
    
    if (allRankings.value.length > 0) {
      ElMessage.success(`加载成功，共 ${allRankings.value.length} 个项目`);
    } else {
      ElMessage.info(`当前阶段暂无评分数据`);
    }
  } catch (error) {
    console.error('加载失败:', error);
    ElMessage.error('加载失败，请稍后重试');
  } finally {
    loading.value = false;
  }
}

function applyFilters() {
  ElMessage.success(`已应用筛选条件`);
}

async function viewDetail(registrationId) {
  selectedDetail.value = allRankings.value.find(
    item => item.registrationId === registrationId
  );
  detailDialogVisible.value = true;
  detailLoading.value = true;
  activeTab.value = currentStage.value;

  try {
    const response = await axios.get(
      `/api/registrations/${registrationId}/review-details`,
      {
        headers: {
          Authorization: `Bearer ${localStorage.getItem('token')}`
        }
      }
    );

    const allDetails = response.data.data;
    // 只保留书审和面谈的评分
    scoreDetails.value = allDetails.filter(d => d.stage === 'BOOK' || d.stage === 'INTERVIEW');
  } catch (error) {
    console.error('加载详情失败:', error);
    ElMessage.error('加载详情失败');
  } finally {
    detailLoading.value = false;
  }
}

function toggleShortlist(row) {
  const id = row.registrationId;
  if (shortlistedIds.value.has(id)) {
    shortlistedIds.value.delete(id);
    ElMessage.info(`已取消入围标记：${row.projectName}`);
  } else {
    shortlistedIds.value.add(id);
    ElMessage.success(`已标记入围：${row.projectName}`);
  }
}

function isShortlisted(row) {
  return shortlistedIds.value.has(row.registrationId);
}

function getStageName(stage) {
  const map = {
    'BOOK': '书审入围',
    'INTERVIEW': '面谈入围'
  };
  return map[stage] || stage;
}

function getStageDescription() {
  const map = {
    'BOOK': '书审入围：查看书审得分排名，筛选进入面谈的项目',
    'INTERVIEW': '面谈入围：查看面谈得分排名，筛选进入决赛的项目'
  };
  return map[currentStage.value] || '';
}

function getStageTarget() {
  const map = {
    'BOOK': '筛选进入面谈',
    'INTERVIEW': '筛选进入决赛'
  };
  return map[currentStage.value] || '';
}

function getStageTagType(stage) {
  const map = {
    'BOOK': 'primary',
    'INTERVIEW': 'success'
  };
  return map[stage] || '';
}

function getGroupTypeLabel(groupType) {
  const map = {
    'BASIC': '基层组',
    'ADVANCED': '进阶组',
    'COMPREHENSIVE': '综合组'
  };
  return map[groupType] || groupType;
}

function getScoreColor(score) {
  if (score >= 90) return '#67c23a';
  if (score >= 80) return '#e6a23c';
  if (score >= 70) return '#f56c6c';
  return '#909399';
}

function getRankTagType(rank) {
  if (rank === 1) return 'danger';
  if (rank === 2) return 'warning';
  if (rank === 3) return 'success';
  return '';
}

function getStatusLabel(row) {
  if (isShortlisted(row)) return '已标记';
  
  const minScore = filters.value.minScore || 0;
  const limit = filters.value.limit || 10;

  if (row.avgTotal >= minScore && row.rank <= limit) {
    return '入围';
  } else if (row.avgTotal >= minScore - 5 && row.avgTotal < minScore) {
    return '待定';
  }
  return '未入围';
}

function getStatusTagType(row) {
  if (isShortlisted(row)) return 'success';
  
  const status = getStatusLabel(row);
  if (status === '入围') return 'success';
  if (status === '待定') return 'warning';
  return 'info';
}

function getRowClassName({ row }) {
  if (isShortlisted(row)) return 'shortlisted-row';
  
  const status = getStatusLabel(row);
  if (status === '入围') return 'qualified-row';
  if (status === '待定') return 'pending-row';
  return '';
}

function exportList() {
  const csvContent = [
    ['阶段', '目标', '排名', '项目名称', '医疗机构', '组别', '平均分', '状态'].join(','),
    ...filteredList.value.map(row => [
      getStageName(currentStage.value),
      getStageTarget(),
      row.rank,
      row.projectName,
      row.institutionName,
      getGroupTypeLabel(row.groupType),
      row.avgTotal.toFixed(1),
      getStatusLabel(row)
    ].join(','))
  ].join('\n');

  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
