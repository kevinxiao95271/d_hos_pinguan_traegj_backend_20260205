# 入围管理页面前端开发指引

> 更新日期：2026-02-07  
> 页面：入围管理  
> 角色：组委会/管理员

---

## 📋 页面功能概述

入围管理页面用于查看书审得分排名，设置入围分数线，筛选入围项目。

### 核心功能

1. ✅ 查看书审得分排名
2. ✅ 按组别筛选
3. ✅ 设置入围分数线
4. ✅ 设置入围数量限制
5. ✅ 查看项目详情
6. ✅ 导出入围名单

---

## 🔌 API 端点

### 1. 获取评分排名

```
GET /api/admin/reviews/rankings
```

**参数：**
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| competitionId | Long | ✅ | 赛事ID |
| stage | String | ✅ | 评审阶段（BOOK=书审） |
| groupType | String | ❌ | 组别筛选（BASIC/ADVANCED/COMPREHENSIVE） |

**返回数据：**

### 2. 获取入围名单（带筛选）

```
GET /api/admin/reviews/shortlist
```

**参数：**
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| competitionId | Long | ✅ | 赛事ID |
| stage | String | ✅ | 评审阶段（BOOK=书审） |
| groupType | String | ❌ | 组别筛选 |
| limit | Integer | ❌ | 入围数量限制（默认10） |
| minAvgTotal | Double | ❌ | 最低分数线 |

**说明：**
- `limit`：前N名入围（如前10名）
- `minAvgTotal`：最低分数线（如85分以上）
- 两个条件可以同时使用（满足分数线的前N名）

**返回数据：**
```json
{
  "code": 200,
  "message": "success",
  "data": [
    {
      "rank": 1,
      "registrationId": 106,
      "projectName": "护理交接班规范化-1",
      "institutionName": "浙江大学医学院附属第一医院",
      "groupType": "BASIC",
      "stage": "BOOK",
      "avgTotal": 92.5
    }
    // ... 最多limit条数据
  ]
}
```

### 颜色标识

- 🟢 **入围项目**：分数 >= 分数线 且 排名 <= 入围数量

- ⚪ **未入围项目**：其他

<script setup>
import { ref, computed, onMounted } from 'vue';
import axios from 'axios';
import { ElMessage } from 'element-plus';

// 筛选条件
const filters = ref({
  groupType: '',
  minScore: null,
  limit: 10
});

// 数据
const allRankings = ref([]);
const loading = ref(false);
const shortlistedIds = ref(new Set());

// 详情对话框
const detailDialogVisible = ref(false);
const selectedDetail = ref(null);
const scoreDetail = ref(null);
const detailLoading = ref(false);

// 计算属性
const filteredList = computed(() => {
  let list = allRankings.value;

  // 按分数筛选
  if (filters.value.minScore) {
    list = list.filter(item => item.avgTotal >= filters.value.minScore);
  }

  // 按数量限制
  if (filters.value.limit) {
    list = list.slice(0, filters.value.limit);
  }

  return list;
});

const averageScore = computed(() => {
  if (allRankings.value.length === 0) return '0.0';
  const sum = allRankings.value.reduce((acc, item) => acc + item.avgTotal, 0);
  return (sum / allRankings.value.length).toFixed(1);
});

const maxScore = computed(() => {
  if (allRankings.value.length === 0) return '0.0';
  return Math.max(...allRankings.value.map(item => item.avgTotal)).toFixed(1);
});

const shortlistedCount = computed(() => {
  return filteredList.value.filter(item => isShortlisted(item)).length;
});

// 方法
async function loadRankings() {
  loading.value = true;

  try {
    const response = await axios.get('/api/admin/reviews/rankings', {
      params: {
        competitionId: 21,
        stage: 'BOOK',
        groupType: filters.value.groupType || undefined
      },
      headers: {
        Authorization: `Bearer ${localStorage.getItem('token')}`
      }
    });

    allRankings.value = response.data.data;
    ElMessage.success(`加载成功，共 ${allRankings.value.length} 个项目`);
  } catch (error) {
    console.error('加载失败:', error);
    ElMessage.error('加载失败，请稍后重试');
  } finally {
    loading.value = false;
  }
}

function applyFilters() {
  // 筛选条件变化时，不重新加载数据，只过滤显示
  ElMessage.success(`已应用筛选条件`);
}

async function viewDetail(registrationId) {
  selectedDetail.value = allRankings.value.find(
    item => item.registrationId === registrationId
  );
  detailDialogVisible.value = true;
  detailLoading.value = true;

  try {
    const response = await axios.get(
      `/api/registrations/${registrationId}/review-details`,
      {
        headers: {
          Authorization: `Bearer ${localStorage.getItem('token')}`
        }
      }
    );

    const details = response.data.data;
    scoreDetail.value = details.find(d => d.stage === 'BOOK');
  } catch (error) {
    console.error('加载详情失败:', error);
    ElMessage.error('加载详情失败');
  } finally {
    detailLoading.value = false;
  }
}

function toggleShortlist(row) {
  const id = row.registrationId;
  if (shortlistedIds.value.has(id)) {
    shortlistedIds.value.delete(id);
    ElMessage.info(`已取消标记：${row.projectName}`);
  } else {
    shortlistedIds.value.add(id);
    ElMessage.success(`已标记入围：${row.projectName}`);
  }
}

function isShortlisted(row) {
  return shortlistedIds.value.has(row.registrationId);
}

function getGroupTypeLabel(groupType) {
  const map = {
    'BASIC': '基层组',
    'ADVANCED': '进阶组',
    'COMPREHENSIVE': '综合组'
  };
  return map[groupType] || groupType;
}

function getScoreColor(score) {
  if (score >= 90) return '#67c23a';
  if (score >= 80) return '#e6a23c';
  if (score >= 70) return '#f56c6c';
  return '#909399';
}

function getRankTagType(rank) {
  if (rank === 1) return 'danger';
  if (rank === 2) return 'warning';
  if (rank === 3) return 'success';
  return '';
}

function getStatusLabel(row) {
  if (isShortlisted(row)) return '已标记';
  
  const minScore = filters.value.minScore || 0;
  const limit = filters.value.limit || 10;

  if (row.avgTotal >= minScore && row.rank <= limit) {
    return '入围';
  } else if (row.avgTotal >= minScore - 5 && row.avgTotal < minScore) {
    return '待定';
  }
  return '未入围';
}

function getStatusTagType(row) {
  if (isShortlisted(row)) return 'success';
  
  const status = getStatusLabel(row);
  if (status === '入围') return 'success';
  if (status === '待定') return 'warning';
  return 'info';
}

function getRowClassName({ row }) {
  if (isShortlisted(row)) return 'shortlisted-row';
  
  const status = getStatusLabel(row);
  if (status === '入围') return 'qualified-row';
  if (status === '待定') return 'pending-row';
  return '';
}

function exportList() {
  // 导出入围名单（这里简化为下载CSV）
  const csvContent = [
    ['排名', '项目名称', '医疗机构', '组别', '平均分', '状态'].join(','),
    ...filteredList.value.map(row => [
      row.rank,
      row.projectName,
      row.institutionName,
      getGroupTypeLabel(row.groupType),
      row.avgTotal.toFixed(1),
      getStatusLabel(row)
    ].join(','))
  ].join('\n');

  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = `入围名单_${new Date().toISOString().slice(0, 10)}.csv`;
  link.click();

  ElMessage.success('导出成功');
}

// 初始化
onMounted(() => {
  loadRankings();
});
</script>

            查看详情
          </Button>
          <Button
            size="small"
            type={shortlistedIds.has(row.registrationId) ? 'default' : 'primary'}
            onClick={() => toggleShortlist(row)}
            style={{ marginLeft: 8 }}
          >
            {shortlistedIds.has(row.registrationId) ? '取消标记' : '标记入围'}
          </Button>
        </>
      )
    }
  ];

  // 过滤数据
  const filteredList = () => {
    let list = allRankings;

    if (filters.minScore) {
      list = list.filter(item => item.avgTotal >= filters.minScore);
    }

    if (filters.limit) {
      list = list.slice(0, filters.limit);
    }

    return list;
  };

  // 统计信息
  const stats = {
    total: allRankings.length,
    average: allRankings.length > 0
      ? (allRankings.reduce((sum, item) => sum + item.avgTotal, 0) / allRankings.length).toFixed(1)
      : '0.0',
    max: allRankings.length > 0
      ? Math.max(...allRankings.map(item => item.avgTotal)).toFixed(1)
      : '0.0',
    shortlisted: filteredList().filter(item => shortlistedIds.has(item.registrationId)).length
  };

  useEffect(() => {
    loadRankings();
  }, []);

  return (
    <div style={{ padding: 20 }}>
      {/* 筛选条件 */}
      <Card style={{ marginBottom: 20 }}>
        <div style={{ display: 'flex', gap: 10 }}>
          <Select
            style={{ width: 150 }}
            placeholder="选择组别"
            value={filters.groupType}
            onChange={(value) => setFilters({ ...filters, groupType: value })}
            allowClear
          >
            <Select.Option value="BASIC">基层组</Select.Option>
            <Select.Option value="ADVANCED">进阶组</Select.Option>
            <Select.Option value="COMPREHENSIVE">综合组</Select.Option>
          </Select>

          <InputNumber
            style={{ width: 150 }}
            placeholder="最低分数线"
            min={0}
            max={100}
            precision={1}
            value={filters.minScore}
            onChange={(value) => setFilters({ ...filters, minScore: value })}
          />

          <InputNumber
            style={{ width: 150 }}
            placeholder="入围数量"
            min={1}
            max={100}
            value={filters.limit}
            onChange={(value) => setFilters({ ...filters, limit: value })}
          />

          <Button type="primary" onClick={loadRankings} loading={loading}>
            刷新数据
          </Button>
        </div>
      </Card>

      {/* 统计信息 */}
      <Card style={{ marginBottom: 20 }}>
        <div style={{ display: 'flex', gap: 30 }}>
          <div>
            <div style={{ fontSize: 14, color: '#909399' }}>总报名数</div>
            <div style={{ fontSize: 24, fontWeight: 'bold' }}>{stats.total}</div>
          </div>
          <div>
            <div style={{ fontSize: 14, color: '#909399' }}>已评审</div>
            <div style={{ fontSize: 24, fontWeight: 'bold' }}>{stats.total}</div>
          </div>
          <div>
            <div style={{ fontSize: 14, color: '#909399' }}>平均分</div>
            <div style={{ fontSize: 24, fontWeight: 'bold' }}>{stats.average}</div>
          </div>
          <div>
            <div style={{ fontSize: 14, color: '#909399' }}>最高分</div>
            <div style={{ fontSize: 24, fontWeight: 'bold' }}>{stats.max}</div>
          </div>
          <div>
            <div style={{ fontSize: 14, color: '#909399' }}>入围数</div>
            <div style={{ fontSize: 24, fontWeight: 'bold', color: '#67c23a' }}>
              {stats.shortlisted}
            </div>
          </div>
        </div>
      </Card>

      {/* 排名列表 */}
      <Card>
        <Table
          columns={columns}
          dataSource={filteredList()}
          rowKey="registrationId"
          loading={loading}
          bordered
          pagination={{
            pageSize: 20,
            showTotal: (total) => `共 ${total} 条`
          }}
        />
      </Card>

      {/* 详情对话框 */}
      <Modal
        title="项目评分详情"
        open={detailVisible}
        onCancel={() => setDetailVisible(false)}
        width={800}
        footer={null}
      >
        {selectedDetail && (
          <div>
            <h3>{selectedDetail.projectName}</h3>
            <Descriptions column={2} bordered>
              <Descriptions.Item label="排名">{selectedDetail.rank}</Descriptions.Item>
              <Descriptions.Item label="总分">{selectedDetail.avgTotal}</Descriptions.Item>
            </Descriptions>

            {scoreDetail && (
              <>
                <h4 style={{ marginTop: 20 }}>分项得分</h4>
                <Table
                  dataSource={[scoreDetail]}
                  bordered
                  pagination={false}
                  columns={[
                    { title: '计划', dataIndex: 'avgPlan', align: 'center' },
                    { title: '问题', dataIndex: 'avgProblem', align: 'center' },
                    { title: '行动', dataIndex: 'avgAction', align: 'center' },
                    { title: '成效', dataIndex: 'avgSuccess', align: 'center' },
                    { title: '回顾', dataIndex: 'avgReview', align: 'center' },
                    { title: '运作', dataIndex: 'avgOperation', align: 'center' },
                    { title: '展示', dataIndex: 'avgPresentation', align: 'center' }
                  ]}
                />

                <h4 style={{ marginTop: 20 }}>评委意见</h4>
                <div>
                  <h5>亮点：</h5>
                  <ul>
                    {scoreDetail.highlights.map((h, i) => (
                      <li key={i}>{h}</li>
                    ))}
                  </ul>

                  <h5>不足：</h5>
                  <ul>
                    {scoreDetail.weaknesses.map((w, i) => (
                      <li key={i}>{w}</li>
                    ))}
                  </ul>
                </div>
              </>
            )}
          </div>
        )}
      </Modal>
    </div>
  );
};

export default ShortlistManagement;

### 2. 入围状态判断

```javascript
function getStatusLabel(row) {
  // 优先判断是否手动标记
  if (isShortlisted(row)) return '已标记';
  
  const minScore = filters.value.minScore || 0;
  const limit = filters.value.limit || 10;

  // 同时满足分数线和排名条件
  if (row.avgTotal >= minScore && row.rank <= limit) {
    return '入围';
  } else if (row.avgTotal >= minScore - 5 && row.avgTotal < minScore) {
    return '待定';  // 接近分数线
  }
  return '未入围';
}
```

function exportList() {
  const csvContent = [
    ['排名', '项目名称', '医疗机构', '组别', '平均分', '状态'].join(','),
    ...filteredList.value.map(row => [

      row.avgTotal.toFixed(1),

  ].join('\n');

  const blob = new Blob(['\uFEFF' + csvContent], { 
    type: 'text/csv;charset=utf-8;'

  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = `入围名单_${new Date().toISOString().slice(0, 10)}.csv`;
  link.click();
}
